Write Java UDFs:
Convert Names to Uppercase: (nano UpperCaseUDF.java)
import org.apache.hadoop.hive.ql.exec.UDF;
import org.apache.hadoop.io.Text;

public class UpperCaseUDF extends UDF {
    public Text evaluate(Text input) {
        if (input == null) return null;
        return new Text(input.toString().toUpperCase());
    }
}
Check Passing Marks (Pass/Fail): (PassFailUDF.java)
import org.apache.hadoop.hive.ql.exec.UDF;

public class PassFailUDF extends UDF {
    public String evaluate(Integer marks) {
        if (marks == null) return "No Marks";
        return marks >= 50 ? "PASS" : "FAIL";
    }
}
Calculate Age from DOB: (AgeCalculatorUDF.java)
import org.apache.hadoop.hive.ql.exec.UDF;
import java.text.SimpleDateFormat;
import java.util.Date;
import java.util.Calendar;

public class AgeCalculatorUDF extends UDF {
    public Integer evaluate(String dob) {
        if (dob == null) return null;

        try {
            SimpleDateFormat sdf = new SimpleDateFormat("yyyy-MM-dd");
            Date birthDate = sdf.parse(dob);

            Calendar birthCal = Calendar.getInstance();
            birthCal.setTime(birthDate);
            
            Calendar currentCal = Calendar.getInstance();

            int age = currentCal.get(Calendar.YEAR) - birthCal.get(Calendar.YEAR);

            // Adjust age if birthday has not yet occurred this year
            if (currentCal.get(Calendar.DAY_OF_YEAR) < birthCal.get(Calendar.DAY_OF_YEAR)) {
                age--;
            }

            return age;
        } catch (Exception e) {
            return null;  // Return null if parsing fails
        }       }    }
