rr.py
#!/usr/bin/env python
import sys
hot_day = None
cold_day = None
hot_temp = float('-inf')
cold_temp = float('inf')
for line in sys.stdin:
    values = line.strip().split("\t")
    
    if len(values) != 3:
        continue  
    
    try:
        date = values[0]
        max_temp = float(values[1])
        min_temp = float(values[2])

        
        if max_temp > hot_temp:
            hot_temp = max_temp
            hot_day = date


        if min_temp < cold_temp:
            cold_temp = min_temp
            cold_day = date

    except ValueError:
        continue  


if hot_day and cold_day:
    print "%s\t%s" % ("Hottest Day:", hot_day)
    print "%s\t%s" % ("Coldest Day:", cold_day)

check mapper output:
hdfs dfs -cat weather/we.txt | python mm.py | sort | python rr.py

Command to run the output:

hadoop jar /usr/lib/hadoop-mapreduce/hadoop-streaming.jar -input weather/we.txt -output weather/output -mapper "python mm.py" -reducer "python rr.py" -file mm.py -file rr.py
