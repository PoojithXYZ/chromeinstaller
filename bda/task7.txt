cw.py
@outputSchema("cleaned_record: tuple(date:chararray, min_temp:float, max_temp:float)")
def clean_data(date, min_temp, max_temp):
    try:
        if not date or not min_temp or not max_temp:
            return None
        
        min_temp = float(min_temp)
        max_temp = float(max_temp)

        if min_temp != min_temp or max_temp != max_temp:
            return None

        if min_temp < 0 or max_temp < 0:
            return None
        
        if max_temp < min_temp:
            return None

        return (date, min_temp, max_temp)

    except (ValueError, TypeError):
        return None



cw.pig
REGISTER 'cw.py' USING jython AS weather_udf;

weather_data = LOAD 't7/w.txt' USING PigStorage(',')
    AS (date:CHARARRAY, min_temp:CHARARRAY, max_temp:CHARARRAY);

filtered_data = FOREACH weather_data GENERATE weather_udf.clean_data(date, min_temp, max_temp) AS cleaned_record;

valid_data = FILTER filtered_data BY cleaned_record IS NOT NULL;

cleaned_weather = FOREACH valid_data GENERATE cleaned_record.date, cleaned_record.min_temp, cleaned_record.max_temp;

STORE cleaned_weather INTO 't7/c' USING PigStorage(',');

Run the script: pig -x mapreduce cw.pig
