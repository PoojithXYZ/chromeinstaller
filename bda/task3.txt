#!/usr/bin/env python
import sys
word_counts = {}
# Read input from stdin
for line in sys.stdin:
    word, count = line.strip().split("\t", 1)  # Split by tab
    try:
        count = int(count)  # Convert count to integer
        if word in word_counts:
            word_counts[word] += count
        else:
            word_counts[word] = count
    except ValueError:
        pass  # Ignore lines with unexpected format

# Print word counts
for word in sorted(word_counts):
    print "%s\t%d" % (word, word_counts[word])

Write a Map Reduce program to perform matrix multiplication. 

Create a file for mapper
 nano m1.py


write mapper class code:
mapper.py:
#!/usr/bin/env python
import sys

M = 2  # Number of rows in Matrix A
N = 2  # Number of columns in Matrix A (also rows in Matrix B)
P = 2  # Number of columns in Matrix B

for line in sys.stdin:
    values = line.strip().split()
    if len(values) != 4:
        continue  # Skip malformed lines

    matrix, row, col, val = values
    row = int(row)
    col = int(col)
    val = float(val)

    if matrix == "A":
        for k in range(P):  # P is the number of columns in Matrix B
            print "%d,%d\tA,%d,%f" % (row, k, col, val)

    elif matrix == "B":
        for i in range(M):  # M is the number of rows in Matrix A
            print "%d,%d\tB,%d,%f" % (i, col, row, val)

reducer.py
#!/usr/bin/env python
import sys

# Debugging: Check if reducer starts
sys.stderr.write("Reducer started...\n")

# Dictionary to store matrix values
matrix_data = {}

# Read input from mapper
for line in sys.stdin:
    sys.stderr.write("Received line: %s\n" % line)  # Debugging
    line = line.strip()

    if "\t" not in line:
        continue  # Skip malformed lines

    try:
        key, value = line.split("\t")
        i, j = map(int, key.split(","))

        parts = value.split(",")
        if len(parts) != 3:
            continue  # Skip incorrectly formatted lines

        matrix_type = parts[0]
        k = int(parts[1])
        val = float(parts[2])

        if (i, j) not in matrix_data:
            matrix_data[(i, j)] = {"A": {}, "B": {}}

        if matrix_type == "A":
            matrix_data[(i, j)]["A"][k] = val
        elif matrix_type == "B":
            matrix_data[(i, j)]["B"][k] = val

    except Exception as e:
        sys.stderr.write("Error processing line: %s\n" % line)
        sys.stderr.write("Exception: %s\n" % str(e))
        continue  # Skip problematic lines

# Compute matrix multiplication result
for (i, j), values in matrix_data.items():
    result = 0
    a_values = values["A"]
    b_values = values["B"]

    for k in a_values:
        if k in b_values:
            result += a_values[k] * b_values[k]

    print "%d,%d\t%f" % (i, j, result)
