from pyspark import SparkContext

sc = SparkContext("local", "WordCount")

input_path = "hdfs:///t11/dataset.txt"
output_path = "hdfs:///t11/output"

text_file = sc.textFile(input_path)

word_counts = (
    text_file.flatMap(lambda line: line.split(" ")).map(lambda word: (word, 1))   .reduceByKey(lambda a, b: a + b)
)

word_counts.saveAsTextFile(output_path)

sc.stop()
  Run the script: spark-submit wordcount.py
